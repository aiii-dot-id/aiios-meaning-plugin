cmake_minimum_required(VERSION 3.20)
project(meaning_worker CXX)

if(APPLE AND "${CMAKE_OSX_DEPLOYMENT_TARGET}" STREQUAL "")
  message(FATAL_ERROR "CMAKE_OSX_DEPLOYMENT_TARGET is not set: name the oldest macOS this worker is built for")
endif()

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(ORT_INCLUDE "" CACHE PATH "The directory that holds onnxruntime_c_api.h")
set(ORT_LIBRARY "" CACHE FILEPATH "The ONNX Runtime library to link")
set(ORT_EXTRA_LINK "" CACHE STRING "Link options the named library needs beside it (a static build's frameworks, for one)")

if(NOT EXISTS "${ORT_INCLUDE}/onnxruntime_c_api.h")
  message(FATAL_ERROR "ORT_INCLUDE does not hold onnxruntime_c_api.h: '${ORT_INCLUDE}'")
endif()
if(NOT EXISTS "${ORT_LIBRARY}")
  message(FATAL_ERROR "ORT_LIBRARY does not exist: '${ORT_LIBRARY}'")
endif()

find_package(Threads REQUIRED)
add_executable(meaning-worker worker.cc)
target_include_directories(meaning-worker PRIVATE "${ORT_INCLUDE}")
target_link_libraries(meaning-worker PRIVATE "${ORT_LIBRARY}" Threads::Threads)
if(ORT_EXTRA_LINK)
  separate_arguments(_extra NATIVE_COMMAND "${ORT_EXTRA_LINK}")
  target_link_options(meaning-worker PRIVATE ${_extra})
endif()
if(APPLE)
  set_target_properties(meaning-worker PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set_target_properties(meaning-worker PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
  target_link_options(meaning-worker PRIVATE -Wl,--disable-new-dtags)
endif()
if(MINGW)
  target_link_options(meaning-worker PRIVATE -municode)
endif()
if(MSVC)
  target_compile_options(meaning-worker PRIVATE /W4 /utf-8)
else()
  target_compile_options(meaning-worker PRIVATE -Wall -Wextra -Wpedantic)
endif()
